        $(function () {
            var showtimes = $("#showtimes").DataTable();

            // Showtime rail keyboard focus visibility and hover animation for reduced motion
            document.getElementById("keyboard_focus").addEventListener("focusin", function() {
                this.style.cursor = "default";
                this.style.outline = "none";

                var currentFocus = showtimes.$header[0];

                if (currentFocus) {
                    // Hover animation for reduced motion
                    setTimeout(function () {
                        document.getElementById("keyboard_focus").classList.add("cursor-pointer");
                        document.getElementById("keyboard_focus_hover").classList.remove("text-decoration-none");

                        currentFocus.style.backgroundColor = "#f5f2e1";
                    }, 10);
                }
            });

            // Keyboard focus out event for reduced motion (showtime rail)
            document.getElementById("keyboard_focus").addEventListener("focusout", function() {
                this.style.cursor = "text";

                var currentFocus = showtimes.$header[0];

                if (currentFocus) {
                    // Hover animation for reduced motion
                    setTimeout(function () {
                        document.getElementById("keyboard_focus").classList.remove("cursor-pointer");

                        currentFocus.style.backgroundColor = "";
                    }, 10);
                }
            });

            showtimes.on('shown.bs.table', function (e) { // Hover animation for reduced motion on shown rows
                var table = e.relatedTarget && e.relatedTarget.jQuery;

                if (!table) return;

                if (table[0] !== e.target[0]) return;

                // Hover animation for reduced motion on showtime rail
                setTimeout(function () {
                    document.getElementById("keyboard_focus").classList.add("cursor-pointer");

                    var currentFocus = table.$header[0];

                    if (currentFocus) {
                        // Hover animation for reduced motion
                        setTimeout(function () {
                            document.getElementById("keyboard_focus").classList.remove("cursor-pointer");

                            currentFocus.style.backgroundColor = "#f5f2e1";
                        }, 10);
                    }
                }, 10);
            });
